cmake_minimum_required(VERSION 3.24)
project(H3DLSSExporter LANGUAGES CXX)

if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "H3DLSSExporter requires Windows x64 and MSVC 2022 or newer.")
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")

set(DLSS_PLAYER_SOURCE "" CACHE PATH "Pinned 2600th/dlss5-video-player source checkout")
set(DLSS_SDK "" CACHE PATH "Pinned official NVIDIA/DLSS SDK checkout")
set(H3_MEDIA_PIPELINE_SOURCE "" CACHE FILEPATH "Generated MediaPipeline.cpp with the audited MP4 container patch")
if(NOT EXISTS "${DLSS_PLAYER_SOURCE}/src/D3D12Renderer.cpp")
    message(FATAL_ERROR "DLSS_PLAYER_SOURCE is not the staged audited source checkout.")
endif()
if(NOT EXISTS "${DLSS_SDK}/include/nvsdk_ngx.h")
    message(FATAL_ERROR "DLSS_SDK is not the staged official NVIDIA SDK checkout.")
endif()
if(NOT EXISTS "${H3_MEDIA_PIPELINE_SOURCE}")
    message(FATAL_ERROR "H3_MEDIA_PIPELINE_SOURCE was not generated by scripts/build_dlss_exporter.ps1.")
endif()

set(PLAYER_SRC "${DLSS_PLAYER_SOURCE}/src")
add_executable(H3DLSSExporter
    H3DLSSExporter.cpp
    "${PLAYER_SRC}/D3D12Renderer.cpp"
    "${PLAYER_SRC}/DLSSBackend.cpp"
    "${PLAYER_SRC}/TemporalGuides.cpp"
    "${PLAYER_SRC}/OpticalFlowNvof.cpp"
    "${PLAYER_SRC}/VideoDecoder.cpp"
    "${PLAYER_SRC}/MediaSource.cpp"
    "${PLAYER_SRC}/RuntimePolicy.cpp"
    "${H3_MEDIA_PIPELINE_SOURCE}"
    "${PLAYER_SRC}/GpuPreference.cpp"
)
target_compile_definitions(H3DLSSExporter PRIVATE
    UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX DLSS_VIDEO_PLAYER_VERSION="0.24.0"
)
target_compile_options(H3DLSSExporter PRIVATE /utf-8 /W4 /permissive- /EHsc /Zc:__cplusplus /guard:cf /GS)
target_link_options(H3DLSSExporter PRIVATE /GUARD:CF /CETCOMPAT /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA)
target_include_directories(H3DLSSExporter PRIVATE "${PLAYER_SRC}" "${DLSS_SDK}/include")
target_link_directories(H3DLSSExporter PRIVATE "${DLSS_SDK}/lib/Windows_x86_64/x64")
target_link_libraries(H3DLSSExporter PRIVATE
    d3d12 dxgi dxguid d3dcompiler mfplat mfreadwrite mfuuid
    ole32 shell32 shlwapi user32 gdi32 winmm bcrypt nvsdk_ngx_d
)

set(NVOF_SDK "${DLSS_PLAYER_SOURCE}/external/nvof")
if(EXISTS "${NVOF_SDK}/nvOpticalFlowD3D12.h")
    target_include_directories(H3DLSSExporter PRIVATE "${NVOF_SDK}")
    target_compile_definitions(H3DLSSExporter PRIVATE DLSS_VIDEO_PLAYER_HAS_NVOF)
endif()
